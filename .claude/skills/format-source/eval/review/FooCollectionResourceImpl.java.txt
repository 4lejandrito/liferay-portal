package com.liferay.example.internal.resource.v1_0;

import com.liferay.portal.kernel.util.ListUtil;
import com.liferay.portal.vulcan.pagination.Page;
import com.liferay.portal.vulcan.pagination.Pagination;

import java.util.ArrayList;
import java.util.List;

import org.osgi.service.component.annotations.Component;
import org.osgi.service.component.annotations.Reference;

/**
 * @author Example Author
 */
@Component(service = FooCollectionResource.class)
public class FooCollectionResourceImpl extends BaseFooCollectionResourceImpl {

	@Override
	public Page<FooCollection> getSiteFooCollectionsPage(
			Long siteId, Pagination pagination)
		throws Exception {

		return Page.of(
			getFooCollection(siteId), pagination,
			_fooCollectionService.getFooCollectionsCount(siteId));
	}

	private List<FooCollection> getFooCollection(long siteId) throws Exception {
		List<FooCollection> fooCollections = new ArrayList<>();

		for (FooCollection fooCollection :
				_fooCollectionService.getFooCollections(siteId)) {

			if ((fooCollection != null) && !fooCollection.isDraft()) {
				fooCollection.setSiteId(siteId);

				fooCollections.add(fooCollection);
			}
		}

		com.liferay.portal.kernel.util.ListUtil.sort(
			fooCollections, new FooCollectionNameComparator());

		return fooCollections;
	}

	@Reference
	private FooCollectionService _fooCollectionService;

}
