package com.liferay.example.internal.security.permission.contributor;

import com.liferay.depot.constants.DepotRolesConstants;
import com.liferay.depot.role.DepotRolePermission;
import com.liferay.portal.kernel.security.permission.ActionKeys;
import com.liferay.portal.kernel.util.ListUtil;

import java.util.List;

import org.osgi.service.component.annotations.Component;

/**
 * @author Example Author
 */
@Component(service = DepotRolePermissionsContributor.class)
public class FooDepotRolePermissionsContributor
	implements DepotRolePermissionsContributor {

	@Override
	public List<DepotRolePermission> getDepotRolePermissions() {
		return ListUtil.fromArray(
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_ADMINISTRATOR,
				FooConstants.RESOURCE_NAME, FooActionKeys.ADD_FOO_COLLECTION,
				FooActionKeys.ADD_FOO_ENTRY),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_ADMINISTRATOR,
				FooCollection.class.getName(), ActionKeys.DELETE,
				ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_ADMINISTRATOR,
				FooEntry.class.getName(), ActionKeys.DELETE,
				ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_ADMINISTRATOR,
				Foo.class.getName(), ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_CONTENT_REVIEWER,
				FooConstants.RESOURCE_NAME, FooActionKeys.ADD_FOO_COLLECTION,
				FooActionKeys.ADD_FOO_ENTRY),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_CONTENT_REVIEWER,
				FooCollection.class.getName(), ActionKeys.DELETE,
				ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_CONTENT_REVIEWER,
				Foo.class.getName(), ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_CONTENT_REVIEWER,
				FooEntry.class.getName(), ActionKeys.DELETE,
				ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_OWNER,
				FooConstants.RESOURCE_NAME, FooActionKeys.ADD_FOO_COLLECTION,
				FooActionKeys.ADD_FOO_ENTRY),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_OWNER,
				FooCollection.class.getName(), ActionKeys.DELETE,
				ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_OWNER,
				FooEntry.class.getName(), ActionKeys.DELETE,
				ActionKeys.UPDATE),
			new DepotRolePermission(
				DepotRolesConstants.DESIGN_LIBRARY_OWNER,
				Foo.class.getName(), ActionKeys.UPDATE));
	}

}
