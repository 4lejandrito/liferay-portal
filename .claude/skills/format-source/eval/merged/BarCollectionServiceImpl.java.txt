package com.liferay.example.service.impl;

import com.liferay.portal.aop.AopService;
import com.liferay.portal.kernel.exception.PortalException;
import com.liferay.portal.kernel.service.ServiceContext;
import com.liferay.portal.kernel.util.ListUtil;

import java.util.List;

import org.osgi.service.component.annotations.Component;
import org.osgi.service.component.annotations.Reference;

/**
 * @author Example Author
 */
@Component(property = "model.class.name=com.liferay.example.model.BarCollection", service = AopService.class)
public class BarCollectionServiceImpl extends BarCollectionServiceBaseImpl {

	@Override
	public BarCollection addBarCollection(
			long groupId, String name, ServiceContext serviceContext)
		throws PortalException {

		_barCollectionModelResourcePermission.check(
			getPermissionChecker(), groupId, ActionKeys.ADD_BAR_COLLECTION);

		return barCollectionLocalService.addBarCollection(
			getUserId(), groupId, name, serviceContext);
	}

	@Override
	public List<BarCollection> getBarCollections(long groupId)
		throws PortalException {

		List<BarCollection> barCollections =
			barCollectionLocalService.getBarCollections(groupId);

		return ListUtil.filter(
			barCollections, barCollection -> _isVisible(barCollection));
	}

	private boolean _isVisible(BarCollection barCollection) {
		if (barCollection.isDraft()) {
			return false;
		}

		return true;
	}

	@Reference
	private ModelResourcePermission<BarCollection>
		_barCollectionModelResourcePermission;

}
