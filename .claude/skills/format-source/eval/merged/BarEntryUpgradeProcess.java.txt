package com.liferay.example.internal.upgrade.v1_1_0;

import com.liferay.petra.string.StringBundler;
import com.liferay.portal.kernel.upgrade.UpgradeProcess;

import java.sql.PreparedStatement;
import java.sql.ResultSet;

/**
 * @author Example Author
 */
public class BarEntryUpgradeProcess extends UpgradeProcess {

	@Override
	protected void doUpgrade() throws Exception {
		try (PreparedStatement preparedStatement1 = connection.prepareStatement(
				_getSelectSQL());
			PreparedStatement preparedStatement2 =
				AutoBatchPreparedStatementUtil.autoBatch(
					connection.prepareStatement(
						"update BarEntry set headId = ? where barEntryId = ?"));

			ResultSet resultSet = preparedStatement1.executeQuery()) {

			while (resultSet.next()) {
				preparedStatement2.setLong(1, resultSet.getLong("headId"));
				preparedStatement2.setLong(2, resultSet.getLong("barEntryId"));

				preparedStatement2.addBatch();
			}

			preparedStatement2.executeBatch();
		}
	}

	private String _getSelectSQL() {
		StringBundler sb = new StringBundler(3);

		sb.append("select barEntryId, headId from ");
		sb.append(_TABLE_NAME_BAR_ENTRY);
		sb.append(" where headId != 0 order by barEntryId asc");

		return sb.toString();
	}

	private static final String _TABLE_NAME_BAR_ENTRY = "BarEntry";

	private static final String _TABLE_NAME_BAR_ENTRY_VERSION =
		"BarEntryVersion";

}
